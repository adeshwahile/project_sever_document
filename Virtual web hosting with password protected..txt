OS:- RHEL9

Requirement:- 192.168.114.130

step 1:- #####setting up name-based virtual hosting is to set up domain name resolution in the /etc/hosts file#####

# vi /etc/hosts
192.168.114.130 www.adesh.com

step 2:- ############Apache web server is installed on your system#########

# yum info httpd ----------------(package informantion)
# yum search httpd ---------------(package search )
# yum install -y httpd -----------(package install)

step 3 :- #########start httpd service #######

# systemctl enable --now httpd

step 4:- create a directory that will contain the website files inside /var/www/html/www.adesh.com

# mkdir /var/www/html/www.adesh.com
# cd www.adesh.com
# vi index.html ----------(page inside this file)
******this is adesh page***********


step 5:- #########Create a Virtual Host Configuration File#########
###########On RHEL, virtual host configs are usually in this directory /etc/httpd/conf.d/   ##########

# vi /etc/httpd/conf.d/virtualhosting.conf

<VirtualHost 192.168.114.130:80>
   ServerName www.adesh.com
   ServerAlias www.adesh.com
q   DocumentRoot /var/www/html/www.adesh.com
ErrorLog /var/log/http/www.adesh.com-error_log
CustomLog /var/log/http/www.adesh.com-access_log common
</VirtualHost>

#########password protected ##################
<Directory "/var/www/html/">
    AuthType Basic
    AuthName "Restricted Area"
    AuthUserFile /etc/httpd/.htpasswd
    Require valid-user
</Directory>
